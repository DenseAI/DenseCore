1. Get the application URL by running these commands:
{{- $base := index .Values "dense-base" -}}
{{- $ctx := dict "root" . "values" $base -}}
{{- if $base.ingress.enabled }}
  {{- range $host := $base.ingress.hosts }}
  {{- range .paths }}
  http{{ if $base.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
  {{- end }}
{{- else if contains "NodePort" $base.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "dense-base.fullname" $ctx }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" $base.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "dense-base.fullname" $ctx }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "dense-base.fullname" $ctx }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ $base.service.port }}
{{- else if contains "ClusterIP" $base.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "dense-base.name" $ctx }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:8080
{{- end }}

2. To check if the model is ready:
  kubectl --namespace {{ .Release.Namespace }} logs -f deployment/{{ include "dense-base.fullname" $ctx }} -c {{ include "dense-base.name" $ctx }}
